#!/usr/bin/env python3
"""RM T1 试跑打分(#276 预注册 c5711367673):16 片段独立前向,固定种子乱序,不告知配对。

打分机制严格照抄 v1 score_rm.py.txt(= 官方 eth-lre/mathtutorbench
compute_scaffolding_score.py):system 四句官方判分提示(原文不改一字)、
user#1 = Problem/Reference Solution、历史 Teacher→assistant/Student→user、
末条 assistant = 待评回复;num_labels=1、bfloat16、禁 dropout、eval。
T1 增项:固定种子乱序(random.Random(SEED) 洗 16 片段次序,配对关系不进
打分路径)、逐前向 torch.manual_seed(确定性设置)。
用法:/tmp/rmenv/bin/python score_t1.py to_score.jsonl scored.jsonl
一次性脚本,存档 .txt;输出 {uid, score, seconds} 逐行 + 摘要行。
"""
from __future__ import annotations

import argparse
import json
import random
import sys
import time

import torch
from transformers import AutoModelForSequenceClassification, AutoTokenizer

MODEL_DIR = "/Users/linsuchang/Desktop/work/models/Qwen2.5-1.5B-pedagogical-rewardmodel"
SEED = 20260917  # 预注册:固定种子乱序(日期种子,跑后不改)
SYSTEM_PROMPT = (
    "Judge the pedagogical quality of the responses provided by two teachers. "
    "Focus on the quality of the scaffolding guidance, correctness, and actionability "
    "of the feedback through nudges, questions and hints. Do not give high scores "
    "for revealing the full answer."
)


def load() -> tuple:
    tok = AutoTokenizer.from_pretrained(MODEL_DIR, trust_remote_code=True)
    try:
        model = AutoModelForSequenceClassification.from_pretrained(
            MODEL_DIR, torch_dtype=torch.bfloat16, trust_remote_code=True, num_labels=1)
    except TypeError:  # transformers 新版参数改名
        model = AutoModelForSequenceClassification.from_pretrained(
            MODEL_DIR, dtype=torch.bfloat16, trust_remote_code=True, num_labels=1)
    for m in model.modules():
        if isinstance(m, torch.nn.Dropout):
            m.p = 0
    model = model.eval()
    device = "mps" if torch.backends.mps.is_available() else "cpu"
    return tok, model.to(device), device


def build_conversation(rec: dict) -> list[dict]:
    conv = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": "Problem: " + rec["problem"] + "\nReference Solution: "
         + rec["reference_solution"]},
    ]
    for e in rec["history"]:
        conv.append({"role": "assistant" if e["role"] == "teacher" else "user",
                     "content": e["text"]})
    conv.append({"role": "assistant", "content": rec["response"]})
    return conv


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("infile")   # to_score.jsonl:{uid, problem, reference_solution, history, response}
    ap.add_argument("outfile")  # scored.jsonl:{uid, score, seconds}
    args = ap.parse_args()
    recs = [json.loads(line) for line in open(args.infile, encoding="utf-8") if line.strip()]
    order = list(range(len(recs)))
    random.Random(SEED).shuffle(order)  # 固定种子乱序;配对信息(uid)不参与打分路径
    tok, model, device = load()
    print(f"device={device} seed={SEED} n={len(recs)} "
          f"torch={torch.__version__}", file=sys.stderr)
    t0 = time.time()
    with open(args.outfile, "w", encoding="utf-8") as fo:
        for i in order:
            rec = recs[i]
            torch.manual_seed(SEED)  # 确定性设置(逐前向)
            inputs = tok.apply_chat_template(
                build_conversation(rec), tokenize=True, return_tensors="pt",
                return_dict=False).to(device)
            with torch.no_grad():
                score = model(inputs).logits[0][0].item()
            fo.write(json.dumps({"uid": rec["uid"], "score": score,
                                 "seconds": round(time.time() - t0, 2)},
                                ensure_ascii=False) + "\n")
            print(f"{rec['uid']}: {score:+.4f}", file=sys.stderr)
    print(f"done {len(recs)} fragments in {time.time() - t0:.1f}s", file=sys.stderr)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
